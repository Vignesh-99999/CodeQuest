//1
#include <stdio.h>

int main() {
    int num1, num2;
    char c;
    
    scanf("%d %c %d", &num1, &c, &num2);
    
    switch(c) {
        case '+':
            printf("%d",num1 + num2);
            break;
        case '-':
            printf("%d",num1 - num2);
            break;
        case '*':
            printf("%d",num1 * num2);
            break;
        case '/':
            if (num2 != 0)
                printf("%d",num1 / num2);
            else
                printf("Error: Division by zero is not allowed.\n");
            break;
        default:
            printf("Error: Unsupported operation '%c'.\n", c);
            break;
    }
    
    return 0;
}

#include <stdio.h>

void printFibonacci(int n) {
    int first = 0, second = 1, next;

    if (n < 1) return;  // If n is less than 1, exit the function


    for (int i = 0; i < n; i++) {
        if (i <= 1) {
            next = i;  // The first two Fibonacci numbers are 0 and 1
        } else {
            next = first + second;  // Calculate the next Fibonacci number
            first = second;         // Update the first number
            second = next;          // Update the second number
        }
        printf("%d ", next);  // Print the next Fibonacci number
    }
 
}

int main() {
    int n;

    scanf("%d", &n);

    printFibonacci(n);

    return 0;
}
